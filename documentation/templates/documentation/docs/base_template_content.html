{% trans "Parameters" %}

  • width - {% trans "Sidebar width" %}
    • "narrow" - 224px
    • "normal" - 288px ({% trans "default" %})
    • "wide" - 384px
  • static - {% trans "Sidebar behavior" %}
    • True - {% trans "Sticky sidebar alongside content (default)" %}
    • False - {% trans "Drawer that slides in as overlay" %}
  • mobile_behavior - {% trans "Behavior on small screens (only for static sidebars)" %}
    • "hidden" - {% trans "Hidden on mobile (default)" %}
    • "drawer" - {% trans "Toggle button appears, opens as drawer" %}
{% endvbox %} {% vbox gap="m" %}
