{% trans "Setup: Tailwind CLI" %}

{% blocktrans %} To customize the theme, you need the Tailwind compiler. We recommend django-tailwind-cli. {% endblocktrans %}

{% vbox gap="m" %}
{% icon name="command-line" %} {% trans "Quick Setup" %}
  1. {% trans "Install the package:" %} uv add django-tailwind-cli
  2. {% trans "Add to INSTALLED_APPS:" %} "django_tailwind_cli"
  3. {% trans "Copy input.css to your project (see below)" %}
  4. {% trans "Run setup:" %} python manage.py tailwind setup
  5. {% trans "Start dev server:" %} python manage.py tailwind runserver
{% endvbox %}
{# fmt:off #} INSTALLED_APPS = [ # ... "django_tailwind_cli", ] # Point to your custom input.css TAILWIND_CLI_SRC_CSS = BASE_DIR / "my_app/static/css/input.css" TAILWIND_CLI_DIST_CSS = "css/tailwind.css" TAILWIND_CLI_AUTOMATIC_DOWNLOAD = False {# fmt:on #}
{# --- Copying input.css --- #} {% vbox gap="s" %}

{% trans "Copying input.css" %}

{% blocktrans %} Copy the default input.css from the installed Insight UI package to your project's static directory. {% endblocktrans %}

{# fmt:off #} # Find the source file in your virtual environment # Linux/macOS: cp .venv/lib/python3.*/site-packages/insight_ui/utils/input.css \ my_app/static/css/input.css # Windows (PowerShell): copy .venv\Lib\site-packages\insight_ui\utils\input.css ` my_app\static\css\input.css {# fmt:on #}

{% blocktrans %} Then update the @source directive at the top of your copied file to include both your templates and the Insight UI templates: {% endblocktrans %}

{# fmt:off #} /* Your project templates */ @source "../../templates"; /* Insight UI templates (adjust path to your venv) */ /* Linux/macOS: */ @source "../../../.venv/lib/python3.12/site-packages/insight_ui/templates/insight_ui"; /* Windows: */ @source "../../../.venv/Lib/site-packages/insight_ui/templates/insight_ui"; {# fmt:on #}

{% trans "Add a @font-face declaration with the path to your font file (relative to input.css)." %}

{% endvbox %} {% vbox gap="s" %}

{% trans "Update the --font-sans variable in the @theme block to use your font." %}

{# fmt:off #} @theme { --font-sans: "my-brand-font", system-ui, sans-serif; } {# fmt:on #}

{% trans "Using Multiple Fonts" %}

{% blocktrans %} You can define multiple font families for different purposes. Tailwind provides built-in variables for sans-serif, serif, and monospace fonts. You can also define custom font families for specific use cases like headings. {% endblocktrans %}

{# fmt:off #} /* Load multiple fonts */ @font-face { font-family: "brand-sans"; src: url("../font/BrandSans.woff2") format("woff2"); font-weight: 100 900; font-display: swap; } @font-face { font-family: "brand-heading"; src: url("../font/BrandHeading.woff2") format("woff2"); font-weight: 400 700; font-display: swap; } @font-face { font-family: "brand-mono"; src: url("../font/BrandMono.woff2") format("woff2"); font-weight: 400; font-display: swap; } @theme { /* Body text */ --font-sans: "brand-sans", system-ui, sans-serif; /* Code blocks */ --font-mono: "brand-mono", ui-monospace, monospace; /* Custom: Headings (use with font-heading class) */ --font-heading: "brand-heading", serif; } {# fmt:on #}

{% trans "Dark Mode" %}

{% blocktrans %} Every color token has a dark variant. The theme switches automatically based on the data-theme="dark" attribute on the root element. {% endblocktrans %}

{# fmt:off #} @theme { /* Light mode (default) */ --color-insight-bg-base: #FFFFFF; --color-insight-text-primary: #1C1C1E; /* Dark mode variants (applied via @layer base) */ --color-insight-bg-base-dark: #030712; --color-insight-text-primary-dark: #F5F8FC; } @layer base { [data-theme="dark"] { --color-insight-bg-base: var(--color-insight-bg-base-dark); --color-insight-text-primary: var(--color-insight-text-primary-dark); } } {# fmt:on #}
{# --- Activating Dark Mode --- #} {% vbox gap="s" %}

{% trans "Activating Dark Mode" %}

{% blocktrans %} Set the data-theme attribute on the <html> element. You can use JavaScript to toggle it or respect the user's system preference. {% endblocktrans %}

{# fmt:off #} // Toggle dark mode manually function toggleTheme() { const html = document.documentElement; const current = html.dataset.theme; html.dataset.theme = current === "dark" ? "light" : "dark"; localStorage.setItem("theme", html.dataset.theme); } // Respect system preference on load function initTheme() { const saved = localStorage.getItem("theme"); if (saved) { document.documentElement.dataset.theme = saved; } else if (window.matchMedia("(prefers-color-scheme: dark)").matches) { document.documentElement.dataset.theme = "dark"; } } // Listen for system preference changes window.matchMedia("(prefers-color-scheme: dark)") .addEventListener("change", (e) => { if (!localStorage.getItem("theme")) { document.documentElement.dataset.theme = e.matches ? "dark" : "light"; } }); {# fmt:on #}

{% blocktrans %} Insight UI includes a theme_toggle component that handles this automatically. See the component documentation for usage. {% endblocktrans %}
